    92→/** Maps business-rule StoreErrors (Thai messages) to tRPC BAD_REQUEST errors. */
    93→function withStoreErrors<T>(fn: () => T): T {
    94→  try {
    95→    return fn();
    96→  } catch (error) {
    97→    if (error instanceof StoreError) {
    98→      throw new TRPCError({ code: "BAD_REQUEST", message: error.message });
    99→    }
   100→    throw error;
   101→  }
   102→}

   104→/** Same-origin cookie policy: secure only when the request is actually https. */
   105→function isSecureRequest(req: TrpcContext["req"]) {
   106→  if (req.protocol === "https") return true;
   107→  const forwardedProto = req.headers["x-forwarded-proto"];
   108→  if (!forwardedProto) return false;
   109→  const protoList = Array.isArray(forwardedProto) ? forwardedProto : String(forwardedProto).split(",");
   110→  return protoList.some((proto) => proto.trim().toLowerCase() === "https");
   111→}

   113→/** Session-priority actor: a signed-in user overrides client-sent actor/role. */
   114→function resolveActor(ctx: TrpcContext, input: { actor?: string; actorRole?: string }) {
   115→  if (ctx.user) {
   116→    return {
   117→      actor: ctx.user.name ?? "ผู้ใช้ระบบ",
   118→      actorRole: ctx.user.role === "admin" ? (ctx.user.appRole === "dev" ? "ผู้พัฒนา" : "ผู้บริหาร") : "เจ้าหน้าที่",
   119→    };
   120→  }
   121→  return { actor: input.actor, actorRole: input.actorRole };
   122→}

   124→/** Admin gate: session role wins; only anonymous requests fall back to input. */

   135→/** ROUND 40 (ข้อ 32) — แปลง mapping ที่ client ส่ง (อาจมี -1 = ไม่นำเข้า) ให้เป็น undefined ตาม contract ของ importTool */
   136→function sanitizeMapping<T extends Record<string, number | undefined>>(mapping: T): T {
   137→  const cleaned: Record<string, number | undefined> = {};
   138→  for (const [key, value] of Object.entries(mapping)) {
   139→    if (value !== undefined && value >= 0) cleaned[key] = value;
   140→  }
   141→  return cleaned as T;
   142→}
