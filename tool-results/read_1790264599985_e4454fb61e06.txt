   269→/**
   270→ * Lightweight SQLite adapter for local feature testing.
   271→ * Uses Node 22's built-in node:sqlite, so a cloned project needs no native binary build.
   272→ * Production auth/runtime remains on the template's configured database.
   273→ */
   274→export function createSqliteStore(filename = ":memory:"): SqliteDb {
   275→  const db = new DatabaseSync(filename);
   276→  db.exec(`
   277→    PRAGMA foreign_keys = ON;
   278→    CREATE TABLE IF NOT EXISTS members (
   279→      id INTEGER PRIMARY KEY AUTOINCREMENT,
   280→      member_number TEXT NOT NULL UNIQUE,
   281→      first_name TEXT NOT NULL,
   282→      last_name TEXT NOT NULL,
   283→      id_card TEXT,
   284→      phone TEXT,
   285→      type TEXT NOT NULL DEFAULT 'สมาชิก',
   286→      balance REAL NOT NULL DEFAULT 0,
   287→      status TEXT NOT NULL DEFAULT 'active' CHECK (status IN ('active', 'suspended')),
   288→      created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP

   291→    -- และ 'refund' (คืนบัตรเงินสด = ผู้ป่วยคืนบัตร ร้านจ่ายเงินคืนตามยอดคงเหลือ)
   292→    CREATE TABLE IF NOT EXISTS member_transactions (
   293→      id INTEGER PRIMARY KEY AUTOINCREMENT,
   294→      member_number TEXT NOT NULL,
   295→      type TEXT NOT NULL CHECK (type IN ('purchase', 'deposit', 'withdrawal', 'activation', 'refund')),
   296→      detail TEXT NOT NULL,
   297→      amount REAL NOT NULL,
   298→      payment_method TEXT,
   299→      balance_after REAL,
   300→      created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
   301→      FOREIGN KEY (member_number) REFERENCES members(member_number)
   302→    );
   303→    CREATE TABLE IF NOT EXISTS store_settings (
   304→      key TEXT PRIMARY KEY,
   305→      value TEXT NOT NULL
   306→    );
   307→    CREATE TABLE IF NOT EXISTS products (
   308→      id INTEGER PRIMARY KEY AUTOINCREMENT,
   309→      sku TEXT NOT NULL UNIQUE,
   310→      barcode TEXT,
   311→      name TEXT NOT NULL,
   312→      category TEXT NOT NULL DEFAULT 'ของใช้',
   313→      emoji TEXT,
   314→      color TEXT,
   315→      unit TEXT NOT NULL DEFAULT 'ชิ้น',
   316→      price REAL NOT NULL,
   317→      cost REAL NOT NULL DEFAULT 0,
   318→      stock INTEGER NOT NULL DEFAULT 0,
   319→      reorder_point INTEGER NOT NULL DEFAULT 10,

   324→      id INTEGER PRIMARY KEY AUTOINCREMENT,

   330→      received_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP

   332→    CREATE TABLE IF NOT EXISTS sales (
   333→      id INTEGER PRIMARY KEY AUTOINCREMENT,
   334→      order_number TEXT NOT NULL UNIQUE,
   335→      payment_method TEXT NOT NULL,

   337→      subtotal REAL NOT NULL,

   339→      total REAL NOT NULL,
   340→      cashier TEXT,
   341→      created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP

   343→    CREATE TABLE IF NOT EXISTS sale_items (
   344→      id INTEGER PRIMARY KEY AUTOINCREMENT,

   348→      qty INTEGER NOT NULL,
   349→      price REAL NOT NULL,
   350→      cost REAL NOT NULL DEFAULT 0

   419→  // ROUND 36 (ข้อ 31) — บังคับเปลี่ยนรหัสผ่านครั้งแรก: 1 = บัญชีใหม่/รหัสที่ผู้บริหารตั้งให้ ต้องเปลี่ยนก่อนใช้งาน

   703→/** Error thrown when a sell-by-money product carries bad price-tiers JSON. */
   704→const PRICE_TIERS_ERROR = "รูปแบบราคาสั้นๆ (price tiers) ไม่ถูกต้อง";
   705→
   706→/** 0/1 coercion for quickPick — accepts boolean or 0/1 from any caller. */

   709→/**
   710→ * Guard for sell-by-money products (round 31-12): priceTiers must be valid
   711→ * JSON — a non-empty array of { label: string, price: number > 0, qty: number > 0 }.
   712→ */
   713→function assertValidPriceTiers(raw: string | null): void {
   714→  let parsed: unknown;
   715→  try {
   716→    parsed = JSON.parse(raw ?? "");
   717→  } catch {
   718→    throw new StoreError(PRICE_TIERS_ERROR);
   719→  }
   720→  if (!Array.isArray(parsed) || parsed.length === 0) throw new StoreError(PRICE_TIERS_ERROR);
   721→  for (const tier of parsed) {
   722→    if (typeof tier !== "object" || tier === null || Array.isArray(tier)) throw new StoreError(PRICE_TIERS_ERROR);
   723→    const { label, price, qty } = tier as { label?: unknown; price?: unknown; qty?: unknown };
   724→    if (
   725→      typeof label !== "string" ||
   726→      !label.trim() ||
   727→      typeof price !== "number" ||
   728→      !Number.isFinite(price) ||
   729→      price <= 0 ||
   730→      typeof qty !== "number" ||
   731→      !Number.isFinite(qty) ||
   732→      qty <= 0
   733→    ) {
   734→      throw new StoreError(PRICE_TIERS_ERROR);
   735→    }
   736→  }

   744→  if (!(input.price >= 0)) throw new StoreError("ราคาขายต้องเป็นตัวเลข 0 ขึ้นไป");
   745→  if (input.stock !== undefined && (!Number.isInteger(input.stock) || input.stock < 0)) throw new StoreError("จำนวนสต็อกเริ่มต้นไม่ถูกต้อง");
   746→  const sellMode = input.sellMode ?? "unit";
   747→  const quickPick = toQuickPickFlag(input.quickPick);
   748→  const priceTiers = input.priceTiers?.trim() || null;
   749→  if (sellMode === "money") assertValidPriceTiers(priceTiers);

   755→    sku,
   756→    input.barcode?.trim() || null,
   757→    name,
   758→    input.category?.trim() || "ของใช้",
   759→    input.emoji ?? null,
   760→    input.color ?? null,
   761→    input.size?.trim() || null,
   762→    input.imageUrl || null,
   763→    input.unit?.trim() || "ชิ้น",
   764→    round2(input.price),
   765→    round2(input.cost ?? 0),
   766→    input.stock ?? 0,
   767→    input.reorderPoint ?? 10,
   768→    quickPick,
   769→    sellMode,
   770→    priceTiers,
   771→  );

   781→ * ค่าเฉลี่ยลงให้เอง — รายงานกำไรจึงถูกต้องตามของจริงเสมอ

   787→export type ReceiveStockResult = {
   788→  /** ทุกบรรทัดที่รับเข้ารอบนี้ (แต่ละบรรทัด = 1 stock_lot ใช้ Lot เดียวกัน) */
   789→  lots: ReceiveStockLotRow[];
   790→  /** บรรทัดแรก — ผู้เรียกแบบเดิม (บรรทัดเดียว) ยังอ่านได้ตามรูปเก่า */
   791→  lot: ReceiveStockLotRow;
   792→  product: SqliteProduct;
   793→  /** ต้นทุนถัวเฉลี่ยใหม่ที่เก็บในสินค้า (ของเดิมในคลัง + ทุกบรรทัดรอบนี้) */
   794→  averageCost: number;
   795→  /** ต้นทุนเฉลี่ยก่อนรับเข้ารอบนี้ */
   796→  previousCost: number;
   797→};
   798→
   799→export function receiveStockLines(
   800→  db: SqliteDb,
   801→  input: { productId: number; lot?: string; expiryDate?: string | null; lines: (ReceiveStockLine | null | undefined)[] | null | undefined; actor?: string; actorRole?: string },
   802→): ReceiveStockResult {
   803→  const lines = (input.lines ?? []).filter((line): line is ReceiveStockLine => line != null);
   804→  if (lines.length === 0) throw new StoreError("กรุณาระบุจำนวนที่รับเข้าอย่างน้อย 1 บรรทัด");
   805→  for (const line of lines) {
   806→    if (!Number.isInteger(line.qty) || line.qty <= 0) throw new StoreError("จำนวนต้องมากกว่า 0");
   807→    if (line.cost !== undefined && !(line.cost >= 0)) throw new StoreError("ต้นทุนต่อหน่วยต้องเป็นตัวเลข 0 ขึ้นไป");
   808→  }
   809→  const product = getProductById(db, input.productId);
   810→  if (!product || !product.active) throw new StoreError(`ไม่พบสินค้ารหัส ${input.productId} ในระบบ`);
   811→  const parsed = lines.map((line) => ({ qty: line.qty, cost: line.cost === undefined ? undefined : round2(line.cost) }));
   812→  const totalQty = parsed.reduce((sum, line) => sum + line.qty, 0);
   813→  const previousCost = product.cost;
   814→  // บรรทัดที่ไม่ใส่ต้นทุน = ใช้ต้นทุนเฉลี่ยปัจจุบันของสินค้า (พฤติกรรมเดิม)
   815→  const lineValue = parsed.reduce((sum, line) => sum + line.qty * (line.cost ?? previousCost), 0);
   816→  const oldQty = Math.max(0, Math.trunc(product.stock));
   817→  const averageCost = round2((oldQty * previousCost + lineValue) / (oldQty + totalQty));
   818→  const lotLabel = input.lot?.trim() || `LOT-${formatDate()}`;
   819→  const createdAt = formatDateTime();
   820→  return runInTransaction(db, () => {
   821→    const insert = db.prepare(`INSERT INTO stock_lots (product_id, lot, expiry_date, qty, cost, received_at, received_qty) VALUES (?, ?, ?, ?, ?, ?, ?)`);
   822→    const lots = parsed.map((line) => {
   823→      const result = insert.run(input.productId, lotLabel, input.expiryDate || null, line.qty, line.cost ?? previousCost, createdAt, line.qty);
   824→      return { id: Number(result.lastInsertRowid), productId: input.productId, lot: lotLabel, expiry: input.expiryDate || null, qty: line.qty, cost: line.cost ?? previousCost, receivedAt: createdAt };
   825→    });
   826→    db.prepare(`UPDATE products SET stock = stock + ?, cost = ? WHERE id = ?`).run(totalQty, averageCost, input.productId);
   827→    // Log: บรรทัดเดียวไม่ใส่ต้นทุน = ข้อความเดิม · มีต้นทุน/หลายบรรทัด = สรุปทุกบรรทัด + ต้นทุนเฉลี่ยใหม่
   828→    const anyCost = parsed.some((line) => line.cost !== undefined);
   829→    const lineSummary = anyCost ? ` · ${parsed.map((line) => `${line.qty}×${money(line.cost ?? previousCost)}${line.cost === 0 ? " แถม" : ""}`).join(" + ")}` : "";
   830→    const avgSummary = anyCost ? ` · ต้นทุนเฉลี่ย ${money(averageCost)} (เดิม ${money(previousCost)})` : "";
   831→    logAction(db, "รับเข้าสินค้า", input.actor?.trim() || "ระบบ", `${product.name} · Lot ${lotLabel} · +${totalQty} ${product.unit}${lineSummary}${avgSummary}`, "green", input.actorRole ?? null, createdAt);
   832→    return { lots, lot: lots[0], product: getProductById(db, input.productId)!, averageCost, previousCost };
   833→  });
   834→}

   836→export function receiveStock(
   837→  db: SqliteDb,
   838→  productId: number,
   839→  lot: string,
   840→  expiryDate: string | null,
   841→  qty: number,
   842→  cost?: number,
   843→  actor?: string,
   844→  actorRole?: string,
   845→) {
   846→  const result = receiveStockLines(db, { productId, lot, expiryDate, lines: [{ qty, cost }], actor, actorRole });
   847→  // รูปเอาต์พุตเดิม { lot, product } + เผื่อ averageCost ไว้ให้ผู้เรียกรุ่นใหม่
   848→  return { lot: result.lot, product: result.product, averageCost: result.averageCost };
   849→}

   851→export function listStockRows(db: SqliteDb): SqliteStockRow[] {
   852→  const rows = db.prepare(`SELECT l.id, p.id AS productId, p.sku, p.name, p.unit, l.lot, l.expiry_date AS expiry, l.qty, l.cost, p.price, p.stock, p.reorder_point AS reorderPoint, l.received_at AS receivedAt, p.active AS active FROM stock_lots l JOIN products p ON p.id = l.product_id ORDER BY p.sku, l.id`).all() as unknown as (Omit<SqliteStockRow, "status" | "tone">)[];
   853→  const today = startOfToday();
   854→  return rows.map((row) => {
   855→    let status: SqliteStockRow["status"] = "ปกติ";
   856→    let tone: SqliteStockRow["tone"] = "green";
   857→    if (!row.active) {
   858→      status = "ปิดการขาย";
   859→      tone = "purple";
   860→    } else if (row.stock <= row.reorderPoint) {

   865→      if (daysLeft <= 30) {
   866→        status = "ใกล้หมดอายุ";
   867→        tone = "amber";
   868→      }
   869→    }
   870→    return { ...row, status, tone };

   883→export function stockCount(db: SqliteDb, input: StockCountInput) {
   884→  if (!Number.isInteger(input.countedQty) || input.countedQty < 0) throw new StoreError("จำนวนที่นับได้ต้องเป็น 0 ขึ้นไป");

   888→  const note = input.note?.trim();

   893→  const tone = diff < 0 ? "red" : diff > 0 ? "green" : "blue";
   894→  runInTransaction(db, () => {
   895→    db.prepare(`UPDATE products SET stock = ? WHERE id = ?`).run(input.countedQty, input.productId);
   896→    logAction(db, "นับสต็อก", input.actor?.trim() || "เจ้าหน้าที่ร้านค้า", detail, tone, input.actorRole ?? null, formatDateTime());

   899→    productId: product.id,
   900→    sku: product.sku,
   901→    name: product.name,
   902→    unit: product.unit,
   903→    systemQty: product.stock,
   904→    countedQty: input.countedQty,
   905→    diff,
   906→    product: getProductById(db, input.productId)!,
   907→  };
   908→}

   939→/**
   940→ * Reconciles physical counts PER LOT: each listed lot's qty is set to its
   941→ * counted value, then the product's sellable stock is recomputed as the SUM
   942→ * of all its lots (lots not listed keep their current qty). One audit entry
   943→ * per product records every lot old→new plus the product total.
   944→ */
   945→export function stockCountLots(db: SqliteDb, input: StockCountLotsInput): StockCountLotsResult {
   946→  if (!Array.isArray(input.counts) || input.counts.length === 0) {
   947→    throw new StoreError("กรุณาระบุจำนวนที่นับได้อย่างน้อย 1 Lot");
   948→  }
   949→  const seen = new Set<number>();
   950→  for (const entry of input.counts) {
   951→    if (!Number.isInteger(entry.lotId)) throw new StoreError("รหัส Lot ไม่ถูกต้อง");
   952→    if (!Number.isInteger(entry.countedQty) || entry.countedQty < 0) {
   953→      throw new StoreError("จำนวนที่นับได้ต้องเป็น 0 ขึ้นไป");
   954→    }
   955→    if (seen.has(entry.lotId)) throw new StoreError("พบ Lot ซ้ำในรายการนับ กรุณาตรวจสอบอีกครั้ง");
   956→    seen.add(entry.lotId);
   957→  }
   958→
   959→  const lotSelect = `SELECT l.id, l.lot, l.expiry_date AS expiry, l.qty, p.id AS productId, p.sku, p.name, p.unit, p.stock, p.active FROM stock_lots l JOIN products p ON p.id = l.product_id`;
   960→  const lots = input.counts.map((entry) => {
   961→    const lot = db.prepare(`${lotSelect} WHERE l.id = ?`).get(entry.lotId) as
   962→      | { id: number; lot: string; expiry: string | null; qty: number; productId: number; sku: string; name: string; unit: string; stock: number; active: number }
   963→      | undefined;
   964→    if (!lot) throw new StoreError(`ไม่พบ Lot รหัส ${entry.lotId} ในระบบ`);
   965→    return { lot, countedQty: entry.countedQty };
   966→  });
   967→
   968→  // A single count session covers exactly one product (the UI enforces it;
   969→  // the server double-checks so mixed submissions can never corrupt stock).
   970→  const productIds = new Set(lots.map((item) => item.lot.productId));
   971→  if (productIds.size !== 1) {
   972→    throw new StoreError("การนับแยกตาม Lot ทำได้ครั้งละ 1 สินค้า");
   973→  }
   974→  const head = lots[0]!.lot;
   975→  if (!head.active) throw new StoreError(`ไม่พบสินค้ารหัส ${head.productId} ในระบบ`);
   976→
   977→  const note = input.note?.trim();
   978→  return runInTransaction(db, () => {
   979→    const results: StockCountLotResult[] = [];
   980→    for (const { lot, countedQty } of lots) {
   981→      db.prepare(`UPDATE stock_lots SET qty = ? WHERE id = ?`).run(countedQty, lot.id);
   982→      results.push({ lotId: lot.id, lot: lot.lot, expiry: lot.expiry, systemQty: lot.qty, countedQty, diff: countedQty - lot.qty });
   983→    }
   984→    // Recompute sellable stock from every lot of this product.
   985→    const { total } = db.prepare(`SELECT COALESCE(SUM(qty), 0) AS total FROM stock_lots WHERE product_id = ?`).get(head.productId) as { total: number };
   986→    db.prepare(`UPDATE products SET stock = ? WHERE id = ?`).run(total, head.productId);
   987→
   988→    const lotBreakdown = results
   989→      .map((result) => `${result.lot}: ${result.systemQty} → ${result.countedQty}${result.diff !== 0 ? ` (${result.diff > 0 ? "+" : ""}${result.diff})` : ""}`)
   990→      .join(" · ");
   991→    const detail = `${head.name} · รวม ${head.stock} → ${total} ${head.unit} (${total - head.stock}) · ${lotBreakdown}` + (note ? ` · ${note}` : "");
   992→    const tone = total < head.stock ? "red" : total > head.stock ? "green" : "blue";
   993→    logAction(db, "นับสต็อก (แยกล็อต)", input.actor?.trim() || "เจ้าหน้าที่ร้านค้า", detail, tone, input.actorRole ?? null, formatDateTime());
   994→
   995→    return {
   996→      productId: head.productId,
   997→      sku: head.sku,
   998→      name: head.name,
   999→      unit: head.unit,
  1000→      systemQty: head.stock,
  1001→      countedQty: total,
  1002→      diff: total - head.stock,
  1003→      lots: results,
  1004→    };
  1005→  });
  1006→}

  1259→export function bulkUpdateProductPrices(db: SqliteDb, input: BulkUpdatePricesInput): BulkUpdatePricesResult {
  1260→  if (input.items.length === 0) throw new StoreError("กรุณาเลือกรายการที่ต้องการปรับราคาอย่างน้อย 1 รายการ");
  1261→  if (input.items.length > 500) throw new StoreError("ปรับราคาได้ครั้งละไม่เกิน 500 รายการ");
  1262→
  1263→  // 1) ตรวจทุกรายการก่อน — ผิดสักชิ้นเดียวโยนทิ้งทั้งชุด (ไม่มี partial write)
  1264→  const seen = new Set<number>();
  1265→  const prepared: Array<{ product: SqliteProduct; price: number | null; cost: number | null }> = [];
  1266→  for (const item of input.items) {
  1267→    if (seen.has(item.productId)) continue; // ซ้ำ (เช่น client ส่งมาสองแถว) — เอาแถวแรก
  1268→    seen.add(item.productId);
  1269→    const product = getProductById(db, item.productId);
  1270→    if (!product) throw new StoreError(`ไม่พบสินค้ารหัส ${item.productId} ในระบบ`);
  1271→    if (item.price !== undefined && !(item.price >= 0)) throw new StoreError(`ราคาขายของ "${product.name}" ต้องเป็นตัวเลข 0 ขึ้นไป`);
  1272→    if (item.cost !== undefined && item.cost < 0) throw new StoreError(`ต้นทุนของ "${product.name}" ต้องเป็นตัวเลข 0 ขึ้นไป`);
  1273→    const nextPrice = item.price !== undefined ? round2(item.price) : null;
  1274→    const nextCost = item.cost !== undefined ? round2(item.cost) : null;
  1275→    if (nextPrice !== null && nextPrice === product.price && nextCost !== null && nextCost === product.cost) continue;
  1276→    prepared.push({ product, price: nextPrice, cost: nextCost });
  1277→  }
  1278→  const willChange = (row: { product: SqliteProduct; price: number | null; cost: number | null }) =>
  1279→    (row.price !== null && row.price !== row.product.price) || (row.cost !== null && row.cost !== row.product.cost);

  1282→    return { updated: 0, priceChanges: 0, costChanges: 0, skipped: input.items.length, sampleNames: [] };
  1283→  }
  1284→
  1285→  // 2) เขียนทั้งชุดใน transaction เดียว
  1286→  let priceChanges = 0;
  1287→  let costChanges = 0;
  1288→  const sampleNames: string[] = [];

  1290→    for (const row of changing) {
  1291→      const assignments: string[] = [];
  1292→      const values: (string | number | null)[] = [];
  1293→      if (row.price !== null && row.price !== row.product.price) {
  1294→        assignments.push("price = ?");
  1295→        values.push(row.price);
  1296→        priceChanges += 1;
  1297→        if (sampleNames.length < 3) sampleNames.push(row.product.name);
  1298→      }
  1299→      if (row.cost !== null && row.cost !== row.product.cost) {
  1300→        assignments.push("cost = ?");
  1301→        values.push(row.cost);
  1302→        costChanges += 1;
  1303→        if (sampleNames.length < 3 && !sampleNames.includes(row.product.name)) sampleNames.push(row.product.name);
  1304→      }
  1305→      if (assignments.length > 0) {
  1306→        db.prepare(`UPDATE products SET ${assignments.join(", ")} WHERE id = ?`).run(...values, row.product.id);
  1307→      }
  1308→    }
  1309→    const actor = input.actor?.trim() || "เจ้าหน้าที่ร้านค้า";
  1310→    const detail = `${changing.length} รายการ · แก้ราคาขาย ${priceChanges} · แก้ต้นทุน ${costChanges}${sampleNames.length > 0 ? ` · เช่น ${sampleNames.join(", ")}${changing.length > sampleNames.length ? " และอื่นๆ" : ""}` : ""}`;
  1311→    logAction(db, "ปรับราคาหลายรายการ", actor, detail, "blue", input.actorRole ?? null, formatDateTime());

  1392→export type CheckoutItemInput =
  1393→  | { productId: number; qty: number; price?: number; unit?: string; label?: string }
  1394→  | { name: string; price: number; qty: number; unit?: string };

  1396→/**
  1397→ * Deducts sold qty from the product's lots FEFO-style (first-expiry-first-out:
  1398→ * earliest expiry first, undated lots last, then oldest received, then lot id).
  1399→ * Best-effort by design: when the lots cannot cover the whole qty — drift
  1400→ * between products.stock and the lot sums — they are zeroed instead and the
  1401→ * shortfall is returned so checkout can flag it in the audit trail.
  1402→ * products.stock stays the authoritative sellable figure at all times.
  1403→ * Returns the unmet qty (0 when the lots covered the sale).
  1404→ */
  1405→function deductLotsFifo(db: SqliteDb, productId: number, qty: number): number {
  1406→  const tracked = db.prepare(`SELECT COUNT(*) AS count FROM stock_lots WHERE product_id = ?`).get(productId) as { count: number };
  1407→  if (tracked.count === 0) return 0; // product is not lot-tracked — nothing to deduct
  1408→  const lots = db.prepare(
  1409→    `SELECT id, qty FROM stock_lots WHERE product_id = ? AND qty > 0 ORDER BY (expiry_date IS NULL), expiry_date, received_at, id`,
  1410→  ).all(productId) as unknown as { id: number; qty: number }[];
  1411→  let remaining = qty;
  1412→  const updateLot = db.prepare(`UPDATE stock_lots SET qty = qty - ? WHERE id = ?`);
  1413→  for (const lot of lots) {
  1414→    if (remaining <= 0) break;
  1415→    const take = Math.min(lot.qty, remaining);
  1416→    updateLot.run(take, lot.id);
  1417→    remaining -= take;
  1418→  }
  1419→  return remaining;
  1420→}

  1433→export function checkout(db: SqliteDb, input: CheckoutInput): CheckoutReceipt {
  1434→  if (!input.items?.length) throw new StoreError("ยังไม่มีสินค้าในบิล");
  1435→  if (!PAYMENT_METHODS.includes(input.paymentMethod)) throw new StoreError("วิธีชำระเงินไม่ถูกต้อง");
  1436→  // Catalog lines merge by productId+price+label (round 31-12: money-mode
  1437→  // tiers of the same product stay separate lines — น้ำแข็ง ฿1 / ฿5 / ฿10);
  1438→  // special ("รายการพิเศษ") lines merge by lowercased name+price+unit.
  1439→  type MergedLine = { productId: number; qty: number; price?: number; unit?: string; label?: string };
  1440→  const merged = new Map<string, MergedLine>();
  1441→  const specials = new Map<string, { name: string; price: number; qty: number; unit: string }>();
  1442→  for (const item of input.items) {
  1443→    if (!Number.isInteger(item.qty) || item.qty <= 0) throw new StoreError("จำนวนต้องมากกว่า 0");
  1444→    if ("productId" in item) {
  1445→      if (!Number.isInteger(item.productId) || item.productId <= 0) throw new StoreError("รหัสสินค้าไม่ถูกต้อง");
  1446→      const unitPrice = item.price !== undefined ? Math.round(item.price * 10000) / 10000 : undefined;
  1447→      const key = `${item.productId}|${unitPrice ?? ""}|${item.label ?? ""}`;
  1448→      const existing = merged.get(key);
  1449→      merged.set(key, { productId: item.productId, qty: (existing?.qty ?? 0) + item.qty, price: unitPrice, unit: item.unit, label: item.label });
  1450→    } else {
  1451→      const name = item.name.trim();
  1452→      if (!name) throw new StoreError("กรุณาระบุชื่อรายการพิเศษ");
  1453→      if (!(item.price > 0)) throw new StoreError("ราคารายการพิเศษต้องมากกว่า 0");
  1454→      const unit = item.unit?.trim() || "รายการ";
  1455→      const key = `${name.toLowerCase()}|${item.price}|${unit}`;
  1456→      const existing = specials.get(key);
  1457→      specials.set(key, { name, price: item.price, qty: (existing?.qty ?? 0) + item.qty, unit });
  1458→    }
  1459→  }
  1460→  const isWallet = input.paymentMethod === WALLET_PAYMENT;
  1461→  if (isWallet && !input.memberNumber?.trim()) throw new StoreError("กรุณาเลือกสมาชิกก่อนชำระแบบหักบัญชีสมาชิก");
  1462→
  1463→  const products = new Map<number, SqliteProduct>();
  1464→  const qtyByProduct = new Map<number, number>();
  1465→  for (const line of Array.from(merged.values())) {
  1466→    qtyByProduct.set(line.productId, (qtyByProduct.get(line.productId) ?? 0) + line.qty);
  1467→  }
  1468→  for (const [productId, totalQty] of Array.from(qtyByProduct.entries())) {
  1469→    const product = getProductById(db, productId);
  1470→    if (!product || !product.active) throw new StoreError(`ไม่พบสินค้ารหัส ${productId} ในระบบ`);
  1471→    if (product.stock < totalQty) throw new StoreError(`สต็อกไม่พอ · ${product.name} เหลือ ${product.stock} ${product.unit}`);
  1472→    products.set(productId, product);
  1473→  }
  1474→
  1475→  const items: { productId: number | null; name: string; unit: string; qty: number; price: number; cost: number; amount: number }[] = [
  1476→    ...Array.from(merged.values()).map((line) => {
  1477→      const product = products.get(line.productId)!;
  1478→      const unitPrice = line.price ?? product.price;
  1479→      return { productId: line.productId, name: line.label ? `${product.name} · ${line.label}` : product.name, unit: line.unit ?? product.unit, qty: line.qty, price: unitPrice, cost: product.cost, amount: round2(unitPrice * line.qty) };
  1480→    }),
  1481→    ...Array.from(specials.values()).map((special) => ({ ...special, productId: null, cost: 0, amount: round2(special.price * special.qty) })),
  1482→  ];
  1483→  const subtotal = round2(items.reduce((sum, item) => sum + item.amount, 0));
  1484→  const discount = round2(Math.max(0, input.discount ?? 0));
  1485→  if (discount > subtotal) throw new StoreError("ส่วนลดมากกว่ายอดรวมสินค้า");
  1486→  const total = round2(subtotal - discount);
  1487→
  1488→  let member: SqliteMember | undefined;
  1489→  if (isWallet) {
  1490→    member = db.prepare(`SELECT id, member_number as memberNumber, prename, first_name as firstName, last_name as lastName, id_card as idCard, phone, type, balance, status, created_at as createdAt, photo_url as photoUrl, gender, credit_limit as creditLimit FROM members WHERE member_number = ?`).get(input.memberNumber!.trim()) as SqliteMember | undefined;
  1491→    if (!member) throw new StoreError(`ไม่พบสมาชิก ${input.memberNumber} ในระบบ`);
  1492→    if (member.status !== "active") throw new StoreError(`สมาชิกถูกระงับการใช้งาน · ${member.memberNumber}`);
  1493→    if (member.balance < total) throw new StoreError(`ยอดเงินคงเหลือไม่พอ · ยอดชำระ ${money(total)} คงเหลือ ${money(member.balance)}`);
  1494→    // Req 23.2 — วงเงินซื้อสินค้าต่อวัน: wallet bills of TODAY plus this bill
  1495→    // must stay within the member's credit limit (null = ไม่จำกัด).
  1496→    if (member.creditLimit != null) {
  1497→      const todayStart = `${formatDate(new Date())} 00:00:00`;
  1498→      const usedRow = db.prepare(`SELECT ROUND(COALESCE(SUM(total), 0), 2) AS usedToday FROM sales WHERE member_number = ? AND payment_method = ? AND created_at >= ?`).get(member.memberNumber, WALLET_PAYMENT, todayStart) as { usedToday: number };
  1499→      const usedToday = round2(usedRow.usedToday ?? 0);
  1500→      if (round2(usedToday + total) > member.creditLimit) {
  1501→        throw new StoreError(`เกินวงเงินซื้อสินค้าต่อวัน · วงเงิน ${money(member.creditLimit)} · ใช้ไปแล้ววันนี้ ${money(usedToday)} · บิลนี้ ${money(total)}`);
  1502→      }
  1503→    }
  1504→  } else if (input.memberNumber?.trim()) {
  1505→    // Cash / QR bills can still be tied to a member — the receipt then shows
  1506→    // ผู้ทำรายการ + ยอดคงเหลือล่าสุด of that member (no deduction happens).
  1507→    member = db.prepare(`SELECT id, member_number as memberNumber, prename, first_name as firstName, last_name as lastName, id_card as idCard, phone, type, balance, status, created_at as createdAt, photo_url as photoUrl, gender, credit_limit as creditLimit FROM members WHERE member_number = ?`).get(input.memberNumber.trim()) as SqliteMember | undefined;
  1508→  }
  1509→
  1510→  const now = new Date();
  1511→  const createdAt = formatDateTime(now);
  1512→  const orderNumber = nextOrderNumber(db, now);
  1513→  const cashier = input.cashier?.trim() || "เจ้าหน้าที่ร้านค้า";
  1514→  // ROUND 63 — แนบหลักฐานการชำระไปกับบิล (ส่วนใหญ่คือสลิป QR / สแกน) · จำกัดความยาวกัน URL ผิดปกติ
  1515→  const slipImage = typeof input.slipImage === "string" && input.slipImage.startsWith("data:image/") ? input.slipImage.slice(0, 600_000) : null;

  1522→      subtotal,
  1523→      discount,
  1524→      total,
  1525→      cashier,
  1526→      createdAt,
  1527→      slipImage,
  1528→    );
  1529→    const saleId = Number(saleResult.lastInsertRowid);
  1530→    const insertItem = db.prepare(`INSERT INTO sale_items (sale_id, product_id, name, qty, price, cost, unit) VALUES (?, ?, ?, ?, ?, ?, ?)`);
  1531→    const updateStock = db.prepare(`UPDATE products SET stock = stock - ? WHERE id = ?`);
  1532→    const lotShortfalls: { name: string; unit: string; qty: number }[] = [];
  1533→    for (const item of items) {
  1534→      insertItem.run(saleId, item.productId, item.name, item.qty, item.price, item.cost, item.unit);
  1535→      if (item.productId == null) continue; // รายการพิเศษ — no product/stock row behind it
  1536→      updateStock.run(item.qty, item.productId);
  1537→      const shortfall = deductLotsFifo(db, item.productId, item.qty);
  1538→      if (shortfall > 0) lotShortfalls.push({ name: item.name, unit: item.unit, qty: shortfall });
  1539→    }
  1540→
  1541→    let balanceAfter: number | null = null;
  1542→    if (isWallet && member) {
  1543→      balanceAfter = round2(member.balance - total);
  1544→      db.prepare(`UPDATE members SET balance = ? WHERE member_number = ?`).run(balanceAfter, member.memberNumber);
  1545→      db.prepare(`INSERT INTO member_transactions (member_number, type, detail, amount, payment_method, balance_after, actor, created_at) VALUES (?, 'purchase', ?, ?, ?, ?, ?, ?)`).run(
  1546→        member.memberNumber,
  1547→        `บิล #${orderNumber} · ${summarizeItems(items)}`,
  1548→        -total,
  1549→        input.paymentMethod,
  1550→        balanceAfter,
  1551→        cashier,
  1552→        createdAt,
  1553→      );
  1554→    } else if (member) {
  1555→      // Cash / QR member bill — snapshot the untouched balance for the slip.
  1556→      balanceAfter = round2(member.balance);
  1557→    }
  1558→
  1559→    const lotNote = lotShortfalls.length
  1560→      ? ` · ตัดยอด Lot ไม่ครบ: ${lotShortfalls.map((row) => `${row.name} ${row.qty} ${row.unit}`).join(", ")} (แนะนำนับสต็อกแยกล็อต)`
  1561→      : "";
  1562→    logAction(db, "ขายสินค้า", cashier, `บิล #${orderNumber} · ${money(total)}${member ? ` · หักบัญชี ${member.memberNumber}` : ""}${slipImage ? " · แนบหลักฐานการชำระแล้ว" : ""}${lotNote}`, "blue", input.actorRole ?? null, createdAt);
  1563→
  1564→    return {
  1565→      orderNumber,
  1566→      createdAt,
  1567→      items: items.map(({ productId, name, unit, qty, price, amount }) => ({ productId, name, unit, qty, price, amount })),
  1568→      subtotal,
  1569→      discount,
  1570→      total,
  1571→      paymentMethod: input.paymentMethod,
  1572→      memberNumber: member?.memberNumber ?? null,
  1573→      memberName: member ? [member.prename, member.firstName, member.lastName].filter(Boolean).join(" ") : null,
  1574→      balanceAfter,
  1575→      cashier,
  1576→    };
  1577→  });
  1578→}

  1814→export function walletTx(db: SqliteDb, input: WalletTxInput) {
  1815→  if (!(input.amount > 0)) throw new StoreError("จำนวนเงินต้องมากกว่า 0");
  1816→  const amount = round2(input.amount);
  1817→  const member = db.prepare(`SELECT id, member_number as memberNumber, prename, first_name as firstName, last_name as lastName, id_card as idCard, phone, type, balance, status, created_at as createdAt, credit_limit as creditLimit, card_number as cardNumber, patient_code as patientCode FROM members WHERE member_number = ?`).get(input.memberNumber?.trim()) as SqliteMember | undefined;
  1818→  if (!member) throw new StoreError(`ไม่พบสมาชิก ${input.memberNumber} ในระบบ`);

  1821→    throw new StoreError(`ยอดเงินคงเหลือไม่พอ · ${input.type === "refund" ? "คืนบัตร" : "ถอน"} ${money(amount)} คงเหลือ ${money(member.balance)}`);
  1822→  }

  1832→  const actor = input.actor?.trim() || "เจ้าหน้าที่ร้านค้า";
  1833→  const createdAt = formatDateTime();
  1834→  // ROUND 63 — รูปหลักฐานที่แนบมากับการเติม/ถอน (เช่น สลิปโอนเข้า) เก็บไปกับรายการ

  1845→      createdAt,

  1851→      actor,
