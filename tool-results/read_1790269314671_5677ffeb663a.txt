    17→/** Round 34 — compact baht for on-chart chips: 14900 → "฿14.9k", 1200000 → "฿1.2M", 0 → "฿0". */
    18→function compactBaht(value: number) {
    19→  if (value <= 0) return "฿0";
    20→  const trim = (n: number) => (n >= 100 ? String(Math.round(n)) : n.toFixed(1).replace(/\.0$/, ""));
    21→  if (value >= 1_000_000) return `฿${trim(value / 1_000_000)}M`;
    22→  if (value >= 1000) return `฿${trim(value / 1000)}k`;
    23→  return `฿${Math.round(value)}`;
    24→}

    26→/**
    27→ * Round 34 — smooth monotone cubic path (Fritsch–Carlson). Returns an SVG "M … C …"
    28→ * path through every point that never overshoots past its neighbours, so flat
    29→ * zero-days can't fake a dip below the data the way a jagged polyline does.
    30→ */
    31→function buildSmoothPath(points: { x: number; y: number }[]) {
    32→  const n = points.length;
    33→  if (n === 0) return "";
    34→  if (n === 1) return `M ${points[0].x.toFixed(2)},${points[0].y.toFixed(2)}`;
    35→  const slopes: number[] = [];
    36→  for (let i = 0; i < n - 1; i += 1) {
    37→    const dx = points[i + 1].x - points[i].x;
    38→    slopes.push(dx > 0 ? (points[i + 1].y - points[i].y) / dx : 0);
    39→  }
    40→  const tangents: number[] = [slopes[0]];
    41→  for (let i = 1; i < n - 1; i += 1) {
    42→    tangents.push(slopes[i - 1] * slopes[i] <= 0 ? 0 : (slopes[i - 1] + slopes[i]) / 2);
    43→  }
    44→  tangents.push(slopes[n - 2]);
    45→  for (let i = 0; i < n - 1; i += 1) {
    46→    if (slopes[i] === 0) {
    47→      tangents[i] = 0;
    48→      tangents[i + 1] = 0;
    49→      continue;
    50→    }
    51→    const alpha = tangents[i] / slopes[i];
    52→    const beta = tangents[i + 1] / slopes[i];
    53→    const square = alpha * alpha + beta * beta;
    54→    if (square > 9) {
    55→      const tau = 3 / Math.sqrt(square);
    56→      tangents[i] = tau * alpha * slopes[i];
    57→      tangents[i + 1] = tau * beta * slopes[i];
    58→    }
    59→  }
    60→  let path = `M ${points[0].x.toFixed(2)},${points[0].y.toFixed(2)}`;
    61→  for (let i = 0; i < n - 1; i += 1) {
    62→    const dx = points[i + 1].x - points[i].x;
    63→    path += ` C ${(points[i].x + dx / 3).toFixed(2)},${(points[i].y + (tangents[i] * dx) / 3).toFixed(2)} ${(points[i + 1].x - dx / 3).toFixed(2)},${(points[i + 1].y + (tangents[i + 1] * dx) / 3).toFixed(2)} ${points[i + 1].x.toFixed(2)},${points[i + 1].y.toFixed(2)}`;
    64→  }
    65→  return path;
    66→}

    85→/** Sums "เติมเงินสมาชิก" audit entries per calendar day for the lilac chart line. */
    86→function buildDepositsByDay(logs: { action: string; detail: string; createdAt: string }[]) {
    87→  const map = new Map<string, number>();
    88→  for (const log of logs) {
    89→    if (log.action !== "เติมเงินสมาชิก") continue;
    90→    const date = log.createdAt.slice(0, 10);
    91→    if (!date) continue;
    92→    map.set(date, (map.get(date) ?? 0) + parseBahtFromDetail(log.detail));
    93→  }
    94→  return map;
    95→}
