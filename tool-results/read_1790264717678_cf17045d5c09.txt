   272→   471→function ensureColumn(db: SqliteDb, table: string, column: string, definition: string) {
   273→   472→  const columns = db.prepare(`PRAGMA table_info(${table})`).all() as unknown as { name: string }[];
   274→   473→  if (!columns.some((entry) => entry.name === column)) {
   275→   474→    db.exec(`ALTER TABLE ${table} ADD COLUMN ${column} ${definition}`);
   276→   475→  }
   277→   476→}
