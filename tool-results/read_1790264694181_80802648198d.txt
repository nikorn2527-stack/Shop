   470→/** Migration-safe column add for databases created before a column existed. */
   471→function ensureColumn(db: SqliteDb, table: string, column: string, definition: string) {
   472→  const columns = db.prepare(`PRAGMA table_info(${table})`).all() as unknown as { name: string }[];
   473→  if (!columns.some((entry) => entry.name === column)) {
   474→    db.exec(`ALTER TABLE ${table} ADD COLUMN ${column} ${definition}`);
   475→  }
   476→}

   478→/**
   479→ * ROUND 65 (user) — ขยายชนิดธุรกรรมสมาชิกจาก 3 เป็น 5 ชนิด (เพิ่ม 'activation' เปิดใช้บัตร
   480→ * และ 'refund' คืนบัตรเงินสด) ให้กับฐานข้อมูลที่สร้างตารางไว้ก่อนหน้า — SQLite
   481→ * แก้ CHECK ไม่ได้โดยตรง จึงสร้างตารางใหม่→คัดลอก→สลับชื่อ (idempotent: ถ้าเจอ
   482→ * 'activation' ในคำนิยามเดิมแล้วถือว่าย้ายเสร็จ ข้ามทันที)
   483→ */
   484→function upgradeMemberTransactionsTypes(db: SqliteDb) {
   485→  const row = db.prepare(`SELECT sql FROM sqlite_master WHERE type = 'table' AND name = 'member_transactions'`).get() as { sql: string } | undefined;
   486→  if (!row?.sql || row.sql.includes("'activation'")) return;
   487→  db.exec(`
   488→    CREATE TABLE member_transactions_new (
   489→      id INTEGER PRIMARY KEY AUTOINCREMENT,
   490→      member_number TEXT NOT NULL,
   491→      type TEXT NOT NULL CHECK (type IN ('purchase', 'deposit', 'withdrawal', 'activation', 'refund')),
   492→      detail TEXT NOT NULL,
   493→      amount REAL NOT NULL,
   494→      payment_method TEXT,
   495→      balance_after REAL,
   496→      created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
   497→      FOREIGN KEY (member_number) REFERENCES members(member_number)
   498→    );
   499→  `);
   500→  db.exec(`
   501→    INSERT INTO member_transactions_new (id, member_number, type, detail, amount, payment_method, balance_after, created_at)
   502→      SELECT id, member_number, type, detail, amount, payment_method, balance_after, created_at FROM member_transactions;
   503→    DROP TABLE member_transactions;
   504→    ALTER TABLE member_transactions_new RENAME TO member_transactions;
   505→  `);

   507→  ensureColumn(db, "member_transactions", "balance_after", "REAL");
   508→  ensureColumn(db, "member_transactions", "actor", "TEXT");
   509→  ensureColumn(db, "member_transactions", "slip_url", "TEXT");
   510→}

   732→      qty <= 0
