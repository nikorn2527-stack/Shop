   420→      limitValue.trim() === "" ||
   421→      !Number.isFinite(value) ||
   422→      !Number.isInteger(value) ||
   423→      value <= 0 ||
   424→      value > 1_000_000
   425→    ) {
   426→      setError(
   427→        "วงเงินใหม่ต้องเป็นตัวเลขจำนวนเต็มมากกว่า 0 และไม่เกิน 1,000,000"

   429→      return;
   430→    }

   432→      setError(
   433→        `วงเงินใหม่ต้องน้อยกว่าวงเงินปัจจุบัน (${money(profile.creditLimit)})`

   435→      return;
   436→    }
   437→    setError("");
   438→    setLimitStep("confirm");
   439→  };
   440→  const confirmLimitRequest = () => {
   441→    const value = Number(limitValue);
   442→    if (!Number.isInteger(value) || value <= 0) {
   443→      setLimitStep("form");
   444→      return;
   445→    }
   446→    limitRequestMutation.mutate({
   447→      memberNumber: member.memberNumber,
   448→      requestedLimit: value,
   449→      reason: limitReason.trim() || undefined,
   450→      actor,
   451→      actorRole,
   452→    });
   453→  };
   454→  return (
   455→    <>
   456→      <SectionHeader
   457→        eyebrow="สมาชิกและบัญชี / รายละเอียด"
   458→        title={`รายละเอียด ${displayName}`}
   459→        description="ตรวจสอบประวัติการซื้อสินค้า เติมเงิน และถอนเงินของสมาชิกแต่ละราย"
   460→        action={
   461→          <button className="outline-button" onClick={onBack}>
   462→            <ArrowDownToLine size={16} style={{ transform: "rotate(90deg)" }} />{" "}
